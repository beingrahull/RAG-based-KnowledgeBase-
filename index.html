LLMAnswer • Pydantic strict JSONPython
class LLMAnswer(BaseModel):
  answer: str = Field(..., min_length=1)
  sources: list[int] = Field(..., description="chunk indices 1..K")
  confidence: float = Field(ge=0, le=1)
  @field_validator("sources")
  def check_sources(cls, v):
    if not v: raise ValueError("Need at least one source or fallback")
    return sorted(set(v))
